\begin{equation}
\begin{aligned}
\text{minimize} \quad & \sum_{i=1}^{n} \sum_{j=1}^{m} c_{ij} x_{ij} \\
\text{subject to} \quad & \sum_{j=1}^{m} x_{ij} = a_i, \quad i = 1, \ldots, n, \\
& \sum_{i=1}^{n} x_{ij} \leq b_j, \quad j = 1, \ldots, m, \\
& x_{ij} \geq 0, \quad i = 1, \ldots, n, \quad j = 1, \ldots, m.
\end{aligned}
\end{equation}
